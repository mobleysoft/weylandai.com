(function(G){'use strict';const clamp=(x,a,b)=>Math.max(a,Math.min(b,x)),norm=v=>{const n=Math.hypot(...v)||1;return v.map(x=>x/n)};
function ray(f,u,aspect,x=.5,y=.55){const right=norm([u[1]*f[2]-u[2]*f[1],u[2]*f[0]-u[0]*f[2],u[0]*f[1]-u[1]*f[0]]),up=[f[1]*right[2]-f[2]*right[1],f[2]*right[0]-f[0]*right[2],f[0]*right[1]-f[1]*right[0]];return norm(f.map((v,i)=>v+right[i]*(x*2-1)*aspect*.76+up[i]*(1-y*2)*.76))}
function create(){return{p:[-3,1.65,12],v:[0,0,0],ground:true,jumpWas:false,accumulator:0,clock:0,tool:'folder',phase:'holstered',phaseAt:0,ammo:8,reserve:32,capacity:8,recoil:0,hits:0,enemy:{p:[-3,0,16],state:'approach',timer:0,hp:5,pinUntil:0},lastRay:null}}
function move(s,input,yaw,dt,resolve){s.accumulator+=clamp(dt,0,.1);let n=0;while(s.accumulator>=1/120-1e-10&&n++<12){const h=1/120;s.accumulator-=h;const jump=!!input.jump;if(s.ground&&jump&&!s.jumpWas){s.v[1]=5.4;s.ground=false;}s.jumpWas=jump;let x=Number(input.x)||0,z=Number(input.y)||0;const len=Math.hypot(x,z);if(len>1){x/=len;z/=len;}const wx=Math.sin(yaw)*z+Math.cos(yaw)*x,wz=Math.cos(yaw)*z-Math.sin(yaw)*x,mag=Math.hypot(wx,wz);if(s.ground){const damp=Math.exp(-7*h);s.v[0]*=damp;s.v[2]*=damp;}if(mag){const ux=wx/mag,uz=wz/mag,along=s.v[0]*ux+s.v[2]*uz,headroom=Math.max(0,4.2*mag-along),push=Math.min(headroom,(s.ground?36:10)*h*mag);s.v[0]+=ux*push;s.v[2]+=uz*push;}const speed=Math.hypot(s.v[0],s.v[2]);if(speed>10){s.v[0]*=10/speed;s.v[2]*=10/speed;}if(!s.ground)s.v[1]-=14*h;let py=s.p[1]+s.v[1]*h;const d=[s.v[0]*h,0,s.v[2]*h],r=resolve?.(s.p.slice(),d);if(r){if(r.accepted){s.p[0]=r.position[0];s.p[2]=r.position[2];if(r.hitX)s.v[0]=0;if(r.hitZ)s.v[2]=0;}else{s.v[0]=s.v[2]=0;}}else{s.p[0]+=d[0];s.p[2]+=d[2];}for(const i of[0,2]){const v=clamp(s.p[i],-63,63);if(v!==s.p[i])s.v[i]=0;s.p[i]=v;}if(py<=1.65){py=1.65;s.v[1]=0;s.ground=true;}if(py>6.3){py=6.3;s.v[1]=Math.min(0,s.v[1]);}s.p[1]=py;}return s;}
function switchTool(s){s.phaseAt=s.clock;s.phase=s.tool==='folder'?'equipping':'holstering';s.tool=s.tool==='folder'?'nailer':'folder';s.recoil=0;}
function reload(s){if(s.tool!=='nailer'||!['idle','firing'].includes(s.phase)||s.ammo===s.capacity||s.reserve<=0)return false;s.phase='reload-open';s.phaseAt=s.clock;return true;}
function fire(s,origin,direction,wallDistance=Infinity,backed=false){if(s.tool!=='nailer'||s.phase!=='idle'||s.ammo<=0)return{fired:false};s.ammo--;s.phase='firing';s.phaseAt=s.clock;s.recoil=1;s.lastRay={origin:origin.slice(),direction:direction.slice()};const center=[s.enemy.p[0],1.22,s.enemy.p[2]],to=center.map((v,i)=>v-origin[i]),t=to.reduce((a,v,i)=>a+v*direction[i],0),off=Math.hypot(...to.map((v,i)=>v-direction[i]*t));const hit=t>0&&off<.74&&t<=wallDistance+.35&&s.enemy.state!=='cleared';if(hit){s.enemy.hp--;if(backed?.point)s.enemy.p=backed.point.slice();s.enemy.timer=0;s.enemy.state=s.enemy.hp<=0?'cleared':backed?'pinned':'stagger';s.enemy.pinUntil=s.clock+2.2;}return{fired:true,hit,pinned:hit&&backed,distance:t};}
function tick(s,dt,resolve){s.clock+=clamp(dt,0,.1);s.recoil=Math.max(0,s.recoil-dt*7);const age=s.clock-s.phaseAt;if(s.phase==='equipping'&&age>.4)s.phase='idle';if(s.phase==='holstering'&&age>.3)s.phase='holstered';if(s.phase==='firing'&&age>.17)s.phase='idle';if(s.phase==='reload-open'&&age>.4)s.phase='reload-feed';if(s.phase==='reload-feed'&&age>1.05){const n=Math.min(s.capacity-s.ammo,s.reserve);s.ammo+=n;s.reserve-=n;s.phase='reload-seat';}if(s.phase==='reload-seat'&&age>1.4)s.phase='idle';const e=s.enemy;e.timer+=dt;if(e.state==='cleared')return;if(e.state==='pinned'){if(s.clock>=e.pinUntil){e.state='approach';e.timer=0;}return;}if(e.state==='stagger'){if(e.timer>.38){e.state='approach';e.timer=0;}return;}const dx=s.p[0]-e.p[0],dz=s.p[2]-e.p[2],dist=Math.hypot(dx,dz);if(e.state==='windup'){if(e.timer>.65){if(dist<1.65&&s.p[1]<2.25)s.hits++;e.state='recover';e.timer=0;}return;}if(e.state==='recover'){if(e.timer>.8){e.state='approach';e.timer=0;}return;}if(dist<1.5){e.state='windup';e.timer=0;}else{const d=[dx/dist*dt*.9,0,dz/dist*dt*.9],r=resolve?.([e.p[0],1.65,e.p[2]],d);if(!r||r.accepted){e.p[0]=r?r.position[0]:e.p[0]+d[0];e.p[2]=r?r.position[2]:e.p[2]+d[2];}}}
function pause(s){s.v.fill(0);s.accumulator=0;s.jumpWas=false;s.recoil=0;s.phase=s.tool==='folder'?'holstered':'idle';}
